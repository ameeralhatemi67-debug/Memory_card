FolderSilhouettePath.dart (Parametric Bézier Math) 
import 'package:flutter/material.dart';

Path buildFolderSilhouettePath({
  required double width,
  required double height,
  required double tabWidth, // Dynamically computed from title TextPainter
  double stepHeight = 30.0, // 8px taller head plateau above the pills resting at Y=8px
  double stepSpan = 36.0,
  double cornerRadius = 16.0,
  double alpha = 0.45,
}) {
  final path = Path();
  final r = Radius.circular(cornerRadius);

  path.moveTo(cornerRadius, 0);
  // 1. Top Title Plateau (Fits Text Width)
  path.lineTo(tabWidth, 0);

  // 2. Cubic S-Curve Transition Bridge (C1 Continuity)
  path.cubicTo(
    tabWidth + (alpha * stepSpan), 0,
    tabWidth + ((1.0 - alpha) * stepSpan), stepHeight,
    tabWidth + stepSpan, stepHeight,
  );

  // 3. Lower Shoulder Shelf (Deck Pills Mount Here)
  path.lineTo(width - cornerRadius, stepHeight);
  path.arcToPoint(Offset(width, stepHeight + cornerRadius), radius: r);

  // 4. Right Edge & Bottom Corners
  path.lineTo(width, height - cornerRadius);
  path.arcToPoint(Offset(width - cornerRadius, height), radius: r);
  path.lineTo(cornerRadius, height);
  path.arcToPoint(Offset(0, height - cornerRadius), radius: r);

  // 5. Left Edge & Top Left Corner
  path.lineTo(0, cornerRadius);
  path.arcToPoint(Offset(cornerRadius, 0), radius: r);

  path.close();
  return path;
}
